\begin{frame}{Our Contributions}
  \vspace{0.6em}
  \begin{itemize}
    \setlength{\itemsep}{1.15em}
    \large
    \item \textbf{SMT-Select, the first multimodal framework for SMT algorithm selection,}
      representing problem instances using both formula graphs and
      natural-language descriptions.

    \item \textbf{A systematic investigation of representation and architecture
      design choices.}

    \item \textbf{A thorough evaluation based on SMT-COMP data}
      demonstrates consistent improvements over prior selectors and
      competition-winning solvers.
  \end{itemize}
\end{frame}

    trainable; the pretrained description encoder is frozen.\\[0.15em]
    \(\bullet\)\enspace
    Pairwise classifiers use pair-specific training data; the graph
    encoder and fusion network are shared across all pairs.
  \end{beamercolorbox}

\end{frame}
